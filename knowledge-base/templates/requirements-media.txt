# AI Knowledge Engine — optional MEDIA dependencies (STT + OCR)
# Install only if you need audio/video transcription or image OCR:
#
#   pip install -r requirements-media.txt
#
# Everything here is pip-only and works out of the box on macOS / Windows /
# Linux. In particular, faster-whisper decodes audio via the bundled PyAV
# libraries, so a system ffmpeg is NOT required. This deliberately avoids the
# common macOS failure where Homebrew's ffmpeg is installed but invisible to
# the IDE's PATH (/opt/homebrew/bin).

# --- Speech-to-text (audio + video) ---
# Default backend. No system ffmpeg needed.
faster-whisper>=1.0,<2.0

# Optional alternative STT backend (requires a system 'ffmpeg' on PATH):
# openai-whisper>=20231117

# --- OCR (images / scans / screenshots) ---
# Default backend: ships its own ONNX models + runtime, no system deps.
# Its bundled model reads Chinese and English only; for other languages
# (e.g. Cyrillic) add Tesseract with that language's data (below).
rapidocr-onnxruntime>=1.3,<2.0

# Optional alternative OCR backend. Needs a system 'tesseract' binary plus the
# data of every language in media.ocr.language (English ships by default):
#   Debian/Ubuntu: sudo apt install tesseract-ocr tesseract-ocr-rus   (one package per language)
#   macOS:         brew install tesseract tesseract-lang
#   Windows:       run the UB-Mannheim Tesseract installer and tick the languages
#                  under "Additional language data"
# `python3 scripts/kb_ocr.py --check` lists the installed and missing language data.
# pytesseract>=0.3.8,<1.0
# Pillow>=10.0,<12.0
